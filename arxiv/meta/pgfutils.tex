\newcommand{\newgroupwidth}[2]%
{\expandafter\xdef\csname groupwidth#1\endcsname{#2}}

\def\currentplot{}
\newsavebox{\groupwidthbox}
\makeatletter
\newenvironment{groupwidth}[1]%
{\edef\groupnumber{#1}%
	\@ifundefined{groupwidth\currentplot}{\pgfmathsetlengthmacro{\mywidth}{\linewidth/\groupnumber}}%
	{\expandafter\let\expandafter\mywidth\csname groupwidth\currentplot\endcsname}%
	\begin{lrbox}{\groupwidthbox}%
		\tikzset{/pgfplots/width={\mywidth}}%
		\ignorespaces}%
	{\end{lrbox}%
	\usebox\groupwidthbox
	\@ifundefined{groupwidth\currentplot}%
	{\pgfmathsetlengthmacro{\mywidth}{\mywidth + (\linewidth - \wd\groupwidthbox)/\groupnumber}}{}%
	\tikzifexternalizing{}{%
		\immediate\write\@auxout{\string\newgroupwidth{\currentplot}{\mywidth}}%
	}}
\makeatother
