    select coords between index={#1}{\numexpr#1+2\relax}]
    table[x expr=\coordindex, y=naive40]{\fttable};

    select coords between index={#1}{\numexpr#1+2\relax},
    point meta=\thisrow{bubbletea_over_llmstation},

      precision=1]{\pgfplotspointmeta}$\times$},
    visualization depends on={\thisrow{label_lift} \as \ftlift},
    every node near coord/.append style={font=\tiny,
      text=black, inner sep=1pt,
      shift={(axis direction cs:0,\ftlift)}}]
    table[x expr=\coordindex, y=bubbletea]{\fttable};
}
