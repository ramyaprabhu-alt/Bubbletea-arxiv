\newcommand{\epftbar}[3]{%
  \addplot[forget plot, fill=white, draw=black, densely dashed,
    /pgf/bar shift=#2,
    point meta={(\thisrow{ft_#1_bubbletea}/\thisrow{ft_#1_base}-1)*100},
    nodes near coords={\pgfmathprintnumber[fixed, precision=0,
      print sign]{\pgfplotspointmeta}\%},
    every node near coord/.append style={font=\tiny, inner sep=1pt,
      solid, rotate=90, anchor=west}]
    table[x expr=\coordindex,
      y expr={max(\thisrow{ft_#1_base}, \thisrow{ft_#1_bubbletea})}]{\eptable};
  \addplot[forget plot, fill=ladderE, draw=black, /pgf/bar shift=#2, #3]
    table[x expr=\coordindex, y=ft_#1_bubbletea]{\eptable};
  \addplot[forget plot, fill=ladderB, draw=black, /pgf/bar shift=#2, #3]
    table[x expr=\coordindex,
      y expr={min(\thisrow{ft_#1_base}, \thisrow{ft_#1_bubbletea})}]{\eptable};
}
