% Timeline of one MoE prefill layer on three GPUs, in the style of

% the drawio figure (about 6.9 pt for box labels, 6.3 pt for annotations).
\def\tlmain{\fontsize{7.7}{8.7}\selectfont}
\def\tlsmall{\fontsize{7.0}{8.0}\selectfont}
\definecolor{tlblue}{RGB}{214,234,248}
\tikzset{
  tl-box/.style={draw, rounded corners=1pt, inner sep=0pt, font=\tlmain},
  tl-route/.style={tl-box, fill=tlblue, draw=blue!45!black},
  tl-exp/.style={tl-box, fill=green!12, draw=green!45!black},
  tl-bub/.style={tl-box, fill=red!15, draw=red!55!black},
  tl-hook/.style={tl-box, fill=orange!20, draw=orange!75!black},
  tl-ft/.style={tl-box, fill=blue!10, draw=blue!50!black},
  tl-step/.style={circle, draw, fill=white, inner sep=0.5pt, font=\tlsmall\bfseries},
}
\begin{tikzpicture}[x=0.072cm, y=1cm, font=\tlmain]
% #1 y of the serving lane, #2 end of expert work, #3 number of chunks, #4 GPU.
\newcommand{\tlgpu}[4]{%
  \fill[black!4, rounded corners=3pt] (-18,#1-0.58) rectangle (104,#1+0.46);
  \node[anchor=west, font=\tlmain\bfseries] at (-17,#1+0.17) {GPU #4};
