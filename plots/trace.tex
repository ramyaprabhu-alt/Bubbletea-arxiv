\newcommand{\trarrivalpanel}[2]{%
  \addplot[ybar, fill=black!35, draw=none,
    /pgf/bar width=10, /pgf/bar shift=5]
    table[x=t, y=#1]{\trarrivals};
  \draw[thin, densely dashed] ({rel axis cs:0,0} |- {axis cs:0,#2})
    -- ({rel axis cs:1,0} |- {axis cs:0,#2});
}
\newcommand{\trlatpanel}[1]{%
  \addplot[fill=sysLLMStation, draw=black] table[x expr=\coordindex,
    y expr=\thisrow{llmstation_#1}/\thisrow{alone_#1}]{\trresults};
  \addplot[fill=sysNaive40, draw=black] table[x expr=\coordindex,
    y expr=\thisrow{naive40_#1}/\thisrow{alone_#1}]{\trresults};
  \addplot[fill=sysBubbleTea, draw=black] table[x expr=\coordindex,
    y expr=\thisrow{bubbletea_#1}/\thisrow{alone_#1}]{\trresults};
  \draw[thin, densely dashed] ({rel axis cs:0,0} |- {axis cs:0,1})
    -- ({rel axis cs:1,0} |- {axis cs:0,1});
  \draw[thin, densely dotted] ({rel axis cs:0,0} |- {axis cs:0,1.1})
    -- ({rel axis cs:1,0} |- {axis cs:0,1.1});
}
