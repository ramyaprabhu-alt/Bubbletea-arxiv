% CSV rows: prompt length x load; a panel plots 3 rows (low, medium, high load)
\pgfplotstableread[col sep=comma]{plots/CSVs/sources.csv}\srctable
% #1 = first CSV row of the panel, #2 = panel options
\newcommand{\srcpanel}[2]{%
  \nextgroupplot[#2]
  % the configurations are successive steps towards BubbleTea
  \addplot[fill=ladderA, draw=black,
    select coords between index={#1}{\numexpr#1+2\relax}]
    table[x expr=\coordindex, y=a_plain]{\srctable};
  \addplot[fill=ladderB, draw=black,
    select coords between index={#1}{\numexpr#1+2\relax}]
    table[x expr=\coordindex, y=b_second_chain]{\srctable};
  \addplot[fill=ladderE, draw=black,
    select coords between index={#1}{\numexpr#1+2\relax}]
    table[x expr=\coordindex, y=e_bubbletea]{\srctable};
}
% tick label: request rate, read from the CSV row
% (\protected: pgfplots expands tick labels before typesetting them)
\protected\def\srcticklabel{\pgfmathtruncatemacro{\srcrow}{\tick}%
  \pgfplotstablegetelem{\srcrow}{req_per_s}\of\srctable
  \pgfmathprintnumber[precision=4]{\pgfplotsretval}}
