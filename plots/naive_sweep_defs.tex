% #1 = first CSV row of the panel, #2 = column suffix (empty for samples/s,
% _ttft, _tpot), #3 = panel options
\newcommand{\nspanel}[3]{%
  \nextgroupplot[#3]
  \addplot[fill=sysNaiveCap, draw=black,
    select coords between index={#1}{\numexpr#1+2\relax},
    point meta=\thisrow{naive6#2}, label left,
    % further left when the 40% label (also left) covers this bar
    visualization depends on={\thisrow{naive40#2} \as \nsforty},
    every node near coord/.append style={/utils/exec={%
      \ifdim\nsforty pt>2pt \pgfkeysalso{/tikz/yshift=5pt}\fi}}]
    table[x expr=\coordindex, y=naive6#2]{\nstable};
  \addplot[fill=sysNaive40, draw=black,
    select coords between index={#1}{\numexpr#1+2\relax},
    point meta=\thisrow{naive40#2}, label left]
    table[x expr=\coordindex, y=naive40#2]{\nstable};
  \addplot[fill=sysNaive, draw=black,
    select coords between index={#1}{\numexpr#1+2\relax},
    point meta=\thisrow{naivenone#2},
    % left of the bar (a right label at medium load hits the next group),
    % unless the 40% bar it would cover is tall or the 6% label is there
    visualization depends on={\thisrow{naive40#2} \as \nsforty},
    visualization depends on={\thisrow{naive6#2} \as \nssix},
    every node near coord/.append style={/utils/exec={%
      \ifdim\nsforty pt>1.6pt \pgfkeysalso{/tikz/anchor=north east}%
      \else\ifdim\nssix pt>2pt \pgfkeysalso{/tikz/anchor=north east}%
      \else \pgfkeysalso{/tikz/anchor=south east}\fi\fi}}]
    table[x expr=\coordindex, y=naivenone#2]{\nstable};
}
